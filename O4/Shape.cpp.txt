#include "Shape.h"

Triangle::Triangle(double a, double b, double c) : sides{a, b, c} {}
std::string Triangle::getName() const { return "Треугольник"; }

Quadrilateral::Quadrilateral(double a, double b, double c, double d) : sides{a, b, c, d} {}
std::string Quadrilateral::getName() const { return "Четырёхугольник"; }

Shape* createShape(const std::string& name) 
{
    if (name == "Треугольник") 
{
        return new Triangle(10, 20, 30);
    }
    else if (name == "Четырёхугольник") {
        return new Quadrilateral(10, 20, 30, 40);    }
    return nullptr;
}

void deleteShape(Shape* shape) 
{
    delete shape;
}
